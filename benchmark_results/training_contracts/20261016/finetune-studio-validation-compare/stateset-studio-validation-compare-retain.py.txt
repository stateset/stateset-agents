"""Retain final checkpoint-comparison acceptance and real queued worker evidence."""
from pathlib import Path
import hashlib
import json
import shutil
import subprocess
import xml.etree.ElementTree as ET

root=Path('/home/dom/stateset-agents')
prefix='stateset-studio-validation-compare'
pipeline=json.loads(Path('/tmp',prefix+'-pipeline-receipt.json').read_text())
assert pipeline['passed'] and len(pipeline['stages'])==6
assert all(row['exit_code']==0 for row in pipeline['stages'])
source=Path(pipeline['finished'])
verified=json.loads((source/'verification.json').read_text())
assert verified['passed'] and verified['source_unchanged'] and verified['tests_unchanged']
current={p.relative_to(root/'stateset_agents').as_posix():hashlib.sha256(p.read_bytes()).hexdigest() for p in (root/'stateset_agents').rglob('*.py')}
tests={p.relative_to(root).as_posix():hashlib.sha256(p.read_bytes()).hexdigest() for p in (root/'tests').rglob('*.py')}
assert current==verified['source_sha256'] and tests==verified['test_source_sha256']
assert 'Success: no issues found in 430 source files' in (source/'types.log').read_text()

def count(name):
    parsed=ET.parse(source/name).getroot()
    suites=list(parsed.iter('testsuite'))
    assert not sum(int(s.get(k,0)) for s in suites for k in ['errors','failures','skipped'])
    comparison_cases=[c for c in parsed.iter('testcase') if c.get('classname','').startswith('tests.unit.test_finetune_validation_comparison')]
    assert len(comparison_cases)==45
    return sum(int(s.get('tests',0)) for s in suites)

def receipt(handle):
    value=json.loads(subprocess.check_output(['/home/dom/agent-builds/buildr','show','--json',handle],text=True))
    return {key:value.get(key) for key in ['id','state','exit_code','argv','created','started','finished','wait_seconds','elapsed_seconds','usage','wait_breakdown']}

integration=json.loads(Path('/tmp',prefix+'-real-worker-verification.json').read_text())
assert integration['passed'] and integration['parent_unchanged'] and integration['installed_origin_verified']
job=receipt(integration['acceptance_job'])
worker=receipt(integration['worker_job'])
assert all(j['state']=='succeeded' and j['exit_code']==0 for j in [job,worker])
draft=Path('/tmp',prefix+'-draft.json')
live=Path('/tmp',prefix+'-live.txt')
assert json.loads(draft.read_text())['draft_preserved'] and 'Inside j:' in live.read_text()
summary=dict(passed=True,source_tests=count('regressions.xml'),installed_wheel_tests=count('wheel.xml'),comparison_cases=45,typed_modules=430,queue_job=job['id'],real_worker_queue_job=worker['id'],wheel_sha256=verified['wheel_sha256'],source_and_tests_unchanged=True,wheel_source_matches=True,wheel_case_identities_match_source=True,real_detached_installed_worker_verified=True,real_model_inference=False,real_provider_verified=False,gpu_verified=False,quality_gain_verified=False,paid_run_launched=False,live_herdr_refreshed=True,draft_preserved=True,count_note='Source and installed-wheel counts overlap; scoring transports use synthetic replies through actual record/replay logic.')
out=root/'benchmark_results/training_contracts/20261016/finetune-studio-validation-compare'
out.mkdir(exist_ok=False)
for name,value in [('summary.json',summary),('queue-receipts.json',[job,worker]),('pipeline-receipt.json',pipeline),('real-worker-verification.json',integration)]:
    (out/name).write_text(json.dumps(value,indent=2)+'\n')
for pattern in ['*.log','*.xml']:
    for path in source.glob(pattern):
        shutil.copy2(path,out/path.name)
for name in ['verification.json','prepare-verification.json','frozen-inventory.json','stage-checkpoint.json','wheel-test-configuration.json','wheel-shards.json','reward-audit.json','preflight6-plan.json']:
    shutil.copy2(source/name,out/name)
for suffix in ['format','gates','prepare','wheel','finish','pipeline','retain','smoke']:
    path=Path('/tmp',prefix+'-'+suffix+'.py')
    shutil.copy2(path,out/(path.name+'.txt'))
shutil.copy2(live,out/'herdr-live.txt')
shutil.copy2(draft,out/'herdr-draft.json')
attempts=out/'development-attempts'; attempts.mkdir()
handles=['c8ca94f899a7','d97af838cc49','ca620182f44d','f31c0370dd0e','f024a439bf62','16276bf87b44','e615eca5e7b9','be21eea2842f']
(attempts/'queue-receipts.json').write_text(json.dumps([receipt(h) for h in handles],indent=2)+'\n')
for handle in handles:
    path=Path('/home/dom/.local/state/agent-builds/logs',handle+'.log')
    shutil.copy2(path,attempts/path.name)
shutil.copy2('/tmp/stateset-studio-validation-compare-20261016/types.log',attempts/'f024a439bf62-types.log')
shutil.copy2('/tmp/stateset-studio-validation-compare-before-subset-guard.json',attempts/'before-subset-guard-receipt.json')
shutil.copy2('/tmp/stateset-studio-validation-compare-before-handoff-guard.json',attempts/'before-handoff-guard-receipt.json')
shutil.copy2('/tmp/test_stateset_validation_handoff_binding.py',attempts/'handoff-regression-probe.py.txt')
shutil.copy2('/tmp/stateset-validation-handoff-probe.xml',attempts/'handoff-regression-probe.xml')
(out/'README.md').write_text(f'''# Queued checkpoint validation comparison: acceptance

{summary['source_tests']} targeted source tests and {summary['installed_wheel_tests']}
matching installed-wheel tests passed without skips or failures. Counts overlap;

print(json.dumps({'path':str(out),'summary':summary},indent=2))
