"""Retain comparison acceptance reports and sanitized queue receipts."""
from pathlib import Path
import hashlib
import json
import shutil
import subprocess
import sys
import xml.etree.ElementTree as ET
import zipfile

root = Path('/home/dom/stateset-agents')
source = Path(sys.argv[1])
job = sys.argv[2]
verification = json.loads((source/'verification.json').read_text())
assert verification['passed'] and verification['source_unchanged'] and verification['tests_unchanged']
package = {p.relative_to(root/'stateset_agents').as_posix(): hashlib.sha256(p.read_bytes()).hexdigest() for p in sorted((root/'stateset_agents').rglob('*.py'))}
assert package == verification['source_sha256']
assert 'Success: no issues found in 423 source files' in (source/'types.log').read_text()
with zipfile.ZipFile(next((source/'dist').glob('*.whl'))) as wheel:
    packed = {n.removeprefix('stateset_agents/'): hashlib.sha256(wheel.read(n)).hexdigest() for n in wheel.namelist() if n.startswith('stateset_agents/') and n.endswith('.py')}
    assert packed == package
    for name in ['stateset_agents/evaluation/rubrics/support_rubric_v4.md', 'stateset_agents/config/presets/experiments/support_recovery_preflight6.json']:
        assert wheel.read(name) == (root/name).read_bytes()

def counts(filename):
    suites = list(ET.parse(source/filename).getroot().iter('testsuite'))
    assert not sum(int(s.get('errors', 0))+int(s.get('failures', 0)) for s in suites)
    skipped = sum(int(s.get('skipped', 0)) for s in suites)
    return {'passed': sum(int(s.get('tests', 0)) for s in suites)-skipped, 'skipped': skipped}

def receipt(handle):
    record = json.loads(subprocess.check_output(['/home/dom/agent-builds/buildr', 'show', '--json', handle], text=True))
    return {key: record.get(key) for key in ['id', 'state', 'exit_code', 'argv', 'created', 'started', 'finished', 'wait_seconds', 'elapsed_seconds', 'reason']}

full = receipt(job)
assert full['state'] == 'succeeded' and full['exit_code'] == 0
plan = json.loads((source/'preflight6-plan.json').read_text())
assert not plan['ready_for_paid_execution']
live = Path('/tmp/stateset-studio-comparison-live.txt')
draft = Path('/tmp/stateset-studio-comparison-draft.txt')
summary = dict(passed=True, queue_job=job, source_queue_job='fe4af9f6c1c4', wheel_queue_jobs=['3ef9fff0fa93','bf8ba571fca6','ba607693f84f'], recovered_completed_stages=True, source_regressions=counts('regressions.xml'), wheel_regressions=counts('wheel.xml'), typed_modules=423, source_unchanged=True, test_sources_unchanged=True, wheel_source_matches=True, wheel_profile_matches=True, wheel_rubric_matches=True, wheel_sha256=verification['wheel_sha256'], real_terminal_comparison_verified=True, live_herdr_refreshed=live.exists(), draft_preserved=json.loads(draft.read_text())['draft_preserved'] if draft.exists() else None, queue_wait_seconds=full['wait_seconds'], gpu_verified=False, quality_gain_verified=False, paid_run_launched=False, count_note='Targeted source and wheel counts overlap; provider/worker boundaries use CPU fixtures and mocks.')
out = root/'benchmark_results/training_contracts/20261016/finetune-studio-comparison'
out.mkdir(parents=True, exist_ok=False)
(out/'summary.json').write_text(json.dumps(summary, indent=2)+'\n')
(out/'queue-receipt.json').write_text(json.dumps(full, indent=2)+'\n')
for handle in ['7d02b8c54257', 'f12fc4124664', 'da9c2b64cb32', 'a94c94b9cf6d', 'fe4af9f6c1c4', 'b163151a6298', '44e8dc769dfd', '3ef9fff0fa93', 'bf8ba571fca6', 'ba607693f84f'] + sys.argv[3:]:
    (out/('focused-'+handle+'.json')).write_text(json.dumps(receipt(handle), indent=2)+'\n')
for name in ['verification.json', 'regressions.xml', 'wheel.xml', 'gates.log', 'types.log', 'wheel-origin.log', 'workflow-lint.log', 'wheel-test-configuration.json', 'reward-audit.json', 'preflight6-plan.json', 'studio-snapshot.log', 'free-recovery-plan.log', 'studio-narrow-snapshot.log', 'agent-schema.log', 'agent-models.log', 'agent-validation.log', 'agent-invalid-validation.log', 'recovered-stages.json', 'frozen-inventory.json', 'wheel-shards.json', 'wheel-shard-1.xml', 'wheel-shard-2.xml', 'wheel-shard-3.xml']:
    shutil.copy2(source/name, out/name)
for name in ['stateset-studio-comparison-verify.py', 'stateset-studio-comparison-retain.py', 'stateset-studio-comparison-resume.py', 'stateset-studio-comparison-finish.py', 'stateset-platform-gates.py']:
    shutil.copy2(Path('/tmp')/name, out/(name+'.txt'))
for path, name in [(live, 'herdr-live.txt'), (draft, 'herdr-draft.txt')]:
    if path.exists():
        shutil.copy2(path, out/name)
for previous in sorted(Path('/tmp').glob('stateset-studio-comparison-20261016*')):
    if previous == source or not (previous/'verification.json').exists():
        continue
    attempt = out/'verification-attempts'/previous.name
    attempt.mkdir(parents=True)
    for name in ['verification.json', 'gates.log', 'types.log']:
        if (previous/name).exists():
            shutil.copy2(previous/name, attempt/name)

print(json.dumps({'path': str(out), 'summary': summary}, indent=2))
