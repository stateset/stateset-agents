"""Retain current integrated picker evidence without certifying the queued suite."""
from pathlib import Path
import hashlib
import json
import shutil
import subprocess
import xml.etree.ElementTree as ET

root=Path('/home/dom/stateset-agents'); candidate=Path('/tmp/stateset-evaluation-case-picker-20261016')
source=json.loads((candidate/'candidate-verification.json').read_text())
wheel=json.loads((candidate/'candidate-wheel-verification.json').read_text())
terminal=json.loads((candidate/'terminal-verification.json').read_text())

integration=json.loads((candidate/'canonical-integration.json').read_text())
assert all(record['passed'] for record in [source,wheel,terminal,post,integration])
assert [stage['name'] for stage in source['stages']]==['ruff','tests','types']
assert 'Success: no issues found in 429 source files' in (candidate/'types.log').read_text()
assert all(hashlib.sha256((root/name).read_bytes()).hexdigest()==digest for name,digest in {**wheel['source_sha256'],**wheel['test_source_sha256']}.items())
def cases(name):
    tree=ET.parse(candidate/name).getroot()
    assert all(not sum(int(s.get(key,0)) for key in ['failures','errors','skipped']) for s in tree.iter('testsuite'))
    return {(c.get('classname'),c.get('name')) for c in tree.iter('testcase')}
src=cases('picker-tests.xml'); installed=cases('picker-wheel.xml'); canonical=cases('canonical-picker.xml')
assert src==installed and len(src)==142 and len(canonical)==12
assert canonical=={case for case in src if case[0].rsplit('.',1)[-1]=='test_evaluation_case_picker'}
out=root/'benchmark_results/training_contracts/20261016/finetune-studio-evaluation-picker'
out.mkdir(parents=True,exist_ok=False)
receipts=[]
for handle in ['d72f4e87a399','b1d6d1554573','9df080fce68d','80f844255c06']:
    job=json.loads(subprocess.check_output(['/home/dom/agent-builds/buildr','show','--json',handle],text=True))
    assert job['state']=='succeeded' and job['exit_code']==0
    receipts.append({key:job.get(key) for key in ['id','state','exit_code','argv','created','started','finished','wait_seconds','elapsed_seconds','usage']})
    shutil.copy2(Path('/home/dom/.local/state/agent-builds/logs')/(handle+'.log'),out/(handle+'.log'))
(out/'queue-receipts.json').write_text(json.dumps(receipts,indent=2)+'\n')
summary={'passed':True,'scope':'picker_delta_and_existing_studio_and_evaluation_contracts','source_tests':142,'installed_wheel_tests':142,'post_integration_picker_tests':12,'typed_modules':429,'real_terminal_checks':2,'canonical_integrated':True,'source_matches_verified_candidate':True,'case_identities_match':True,'wheel_sha256':wheel['wheel_sha256'],'full_acceptance_verified':False,'full_acceptance_queue_job':'16e925fa033f','paid_run_launched':False,'gpu_verified':False,'quality_gain_verified':False,'count_note':'Source/wheel counts overlap, and the 12 post-integration cases are a subset.'}
