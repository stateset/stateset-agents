"""Retain immutable studio acceptance evidence after frozen-source verification."""
from pathlib import Path
import hashlib
import json
import shutil
import sys
import subprocess
import xml.etree.ElementTree as ET
import zipfile

root = Path('/home/dom/stateset-agents')
source = Path(sys.argv[1])
job = sys.argv[2]
v = json.loads((source/'verification.json').read_text())
assert v['passed'] and v['source_unchanged'] and v['tests_unchanged']
package = {p.relative_to(root/'stateset_agents').as_posix(): hashlib.sha256(p.read_bytes()).hexdigest() for p in sorted((root/'stateset_agents').rglob('*.py'))}
assert package == v['source_sha256']
assert 'Success: no issues found in 422 source files' in (source/'types.log').read_text()
with zipfile.ZipFile(next((source/'dist').glob('*.whl'))) as wheel:
    packed = {n.removeprefix('stateset_agents/'): hashlib.sha256(wheel.read(n)).hexdigest() for n in wheel.namelist() if n.startswith('stateset_agents/') and n.endswith('.py')}
    assert packed == package
    for name in ['stateset_agents/evaluation/rubrics/support_rubric_v4.md', 'stateset_agents/config/presets/experiments/support_recovery_preflight6.json']:
        assert wheel.read(name) == (root/name).read_bytes()
def counts(name):
    suites = list(ET.parse(source/name).getroot().iter('testsuite'))
    assert not sum(int(s.get('errors', 0))+int(s.get('failures', 0)) for s in suites)
    skipped = sum(int(s.get('skipped', 0)) for s in suites)
    return dict(passed=sum(int(s.get('tests', 0)) for s in suites)-skipped, skipped=skipped)
plan = json.loads((source/'preflight6-plan.json').read_text())
assert not plan['ready_for_paid_execution'] and plan['estimated_total_cost_usd'] is None
assert not (source/'unused-studio-workspace').exists()
live = Path('/tmp/stateset-studio-failures-live.txt')
summary = dict(passed=True, queue_job=job, typed_modules=422, source_unchanged=True, test_sources_unchanged=True, wheel_source_matches=True, wheel_profile_matches=True, wheel_rubric_matches=True, wheel_sha256=v['wheel_sha256'], source_regressions=counts('regressions.xml'), wheel_regressions=counts('wheel.xml'), real_terminal_navigation_verified=True, real_terminal_run_search_verified=True, live_herdr_refreshed=live.exists(), studio_worker_tests='CPU fixtures and mocked CUDA/provider boundaries; no real GPU training', failure_tests='Synthetic exceptions and mocked process/provider boundaries; no actual OOM, disk exhaustion or paid training', distributed_contracts_repeated=False, gpu_verified=False, quality_gain_verified=False, paid_run_launched=False, count_note='Targeted source and wheel counts overlap.')
out = root/'benchmark_results/training_contracts/20261016/finetune-studio-failures'
out.mkdir(parents=True, exist_ok=False)
receipt = json.loads(subprocess.check_output(['/home/dom/agent-builds/buildr', 'show', '--json', job], text=True))
assert receipt['state'] == 'succeeded' and receipt['exit_code'] == 0
safe_receipt = {key: receipt.get(key) for key in ['id', 'state', 'exit_code', 'argv', 'created', 'started', 'finished', 'wait_seconds', 'elapsed_seconds']}
summary['queue_wait_seconds'] = safe_receipt['wait_seconds']
(out/'queue-receipt.json').write_text(json.dumps(safe_receipt, indent=2)+'\n')
(out/'summary.json').write_text(json.dumps(summary, indent=2)+'\n')
for name in ['verification.json', 'regressions.xml', 'wheel.xml', 'gates.log', 'types.log', 'wheel-origin.log', 'workflow-lint.log', 'wheel-test-configuration.json', 'reward-audit.json', 'preflight6-plan.json', 'studio-snapshot.log', 'free-recovery-plan.log', 'studio-narrow-snapshot.log']:
    shutil.copy2(source/name, out/name)
for name in ['stateset-studio-failures-verify.py', 'stateset-studio-failures-retain.py', 'stateset-platform-gates.py']:
    shutil.copy2(Path('/tmp')/name, out/(name+'.txt'))
if live.exists():
    shutil.copy2(live, out/'herdr-live.txt')
readiness = Path('/tmp/stateset-studio-failures-draft.txt')
if readiness.exists():
    shutil.copy2(readiness, out/'herdr-draft.txt')
for previous in sorted(Path('/tmp').glob('stateset-studio-failures-20261016*')):
    if previous == source or not (previous/'verification.json').exists():
        continue
    attempts = out/'verification-attempts'/previous.name
    attempts.mkdir(parents=True)
    for name in ['verification.json', 'gates.log', 'types.log']:
        if (previous/name).exists():
            shutil.copy2(previous/name, attempts/name)
(out/'README.md').write_text(f'''# Fine-tuning failure guidance acceptance

Buildr job `{job}` verified frozen package/test Python and an isolated installed
wheel. Package Python, packaged recovery profile and rubric bytes match.

- {summary['source_regressions']['passed']:,} targeted source regressions passed ({summary['source_regressions']['skipped']} skips).
- {summary['wheel_regressions']['passed']:,} installed-wheel regressions passed ({summary['wheel_regressions']['skipped']} skips).
- All 422 package modules passed typing. Lint, serial formatting, hygiene,
  release governance, stable API compatibility and workflow lint passed.
- Existing real terminal tests cover editing and run search at 74x44/110x32.

Failure contracts cover known CUDA/dependency/memory/storage/timeout failures,
invalid review inputs before worker startup, missing artifacts and completion
receipts, preserving the child's original diagnosis/exit code, corrupt state,
inability to persist a diagnosis, bounded/whitelisted diagnosis reads and
recovery guidance in terminal run details. Arbitrary exception messages and
recorded action/title fields are not copied into structured diagnoses or used
as recovery instructions. Generic failures are not guessed from log strings.

Recovery guidance starts no work. Copying settings remains a fresh reviewed
launch, not checkpoint resume or automatic provider retry. Remote failure
views direct users to inspect active resources/charges before a new paid run.

Failure tests use synthetic exceptions and mocked process/provider boundaries;
no actual OOM, disk exhaustion or paid training was induced. Existing tiny CPU
worker, SFT, reward/judge and recovery tests provide regression coverage. These
are not GPU/provider quality or billing measurements. Broader historical
training/distributed matrices were not repeated. Source/wheel counts overlap.
The installed-wheel recovery planner still blocks paid execution.
Live Herdr refresh/capture: {summary['live_herdr_refreshed']}.
''')
print(json.dumps(dict(path=str(out), summary=summary), indent=2))
