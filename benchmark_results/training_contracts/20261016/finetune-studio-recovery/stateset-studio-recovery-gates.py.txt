"""Repository gates plus serial formatting for this update's changed Python."""
from pathlib import Path
import subprocess
import sys
import black
import isort

for command in [
    [sys.executable,'-m','ruff','check','.'],
    [sys.executable,'scripts/check_repo_hygiene.py'],
    [sys.executable,'scripts/check_release_governance.py'],
    [sys.executable,'scripts/check_api_compatibility.py'],
]:
    print('Checking:', ' '.join(command), flush=True)
    subprocess.run(command,check=True)
for name in ['stateset_agents/finetune_recovery.py', 'stateset_agents/finetune_workspace.py', 'stateset_agents/finetune_worker.py', 'stateset_agents/training/sft.py', 'stateset_agents/tui.py', 'stateset_agents/cli_finetune.py', 'stateset_agents/api/routers/finetune.py', 'tests/unit/test_finetune_recovery.py', 'tests/api/test_finetune_operator.py', 'tests/unit/test_cli_finetune.py','tests/unit/test_mypy_allowlist_ratchet.py']:
    changed=black.format_file_in_place(Path(name),fast=False,mode=black.Mode(line_length=88,target_versions={black.TargetVersion.PY310}),write_back=black.WriteBack.CHECK)
    assert not changed and isort.check_file(name), name
print('Serial formatting passed for all changed Python files.',flush=True)
