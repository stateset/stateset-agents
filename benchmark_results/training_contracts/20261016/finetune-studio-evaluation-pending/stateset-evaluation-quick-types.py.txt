"""Run the read-only type gate independently and retain its exact source identity."""
from pathlib import Path
import hashlib
import json
import os
import subprocess
import sys

root = Path('/home/dom/stateset-agents')
os.chdir(root)
os.environ['PATH'] = str(root/'.venv/bin') + os.pathsep + os.environ['PATH']

def inventory():
    return {p.relative_to(root/'stateset_agents').as_posix(): hashlib.sha256(p.read_bytes()).hexdigest() for p in (root/'stateset_agents').rglob('*.py')}

before = inventory()
result = subprocess.run([sys.executable, 'scripts/check_types.py', '--all'], stdout=subprocess.PIPE, stderr=subprocess.STDOUT, text=True)
print(result.stdout, end='', flush=True)
unchanged = before == inventory()
value = {'passed': result.returncode == 0 and unchanged and 'Success: no issues found in 429 source files' in result.stdout, 'exit_code': result.returncode, 'source_unchanged': unchanged, 'source_sha256': before, 'gate_inputs_sha256': {name:hashlib.sha256((root/name).read_bytes()).hexdigest() for name in ['mypy.ini','scripts/check_types.py']}}
Path('/tmp/stateset-evaluation-quick-types.json').write_text(json.dumps(value,indent=2)+'\n')
raise SystemExit(0 if value['passed'] else 1)
