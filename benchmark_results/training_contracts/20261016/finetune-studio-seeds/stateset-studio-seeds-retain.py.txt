import xml.etree.ElementTree as ET

root = Path('/home/dom/stateset-agents')

assert 'Success: no issues found in 429 source files' in (source/'types.log').read_text()

def count(name):
    suites = list(ET.parse(source/name).getroot().iter('testsuite'))
    assert not sum(int(s.get(k,0)) for s in suites for k in ['errors','failures','skipped'])
    return sum(int(s.get('tests',0)) for s in suites)

def receipt(handle):
    value=json.loads(subprocess.check_output(['/home/dom/agent-builds/buildr','show','--json',handle],text=True))
    return {key:value.get(key) for key in ['id','state','exit_code','argv','created','started','finished','wait_seconds','elapsed_seconds','usage','wait_breakdown']}

job = receipt(sys.argv[1])
assert job['state']=='succeeded' and job['exit_code']==0
live = Path('/tmp', prefix+'-live.txt')

summary = dict(passed=True,source_tests=count('regressions.xml'),installed_wheel_tests=count('wheel.xml'),typed_modules=429,queue_job=job['id'],wheel_sha256=verification['wheel_sha256'],source_and_tests_unchanged=True,wheel_bytes_match_source=True,wheel_case_identities_match_source=True,real_cpu_same_seed_adapter_identity_verified=True,real_cpu_different_seed_adapter_difference_verified=True,training_vs_evaluation_seed_binding_verified=True,provider_seed_boundaries_mocked=True,real_pty_seed_editor_sizes=[[74,44],[110,32]],live_herdr_refreshed=False,draft_preservation_verified_in_this_stage=False,live_judge_verified=False,gpu_verified=False,remote_evaluation_verified=False,quality_gain_verified=False,paid_run_launched=False,count_note='Source and installed-wheel cases overlap; synthetic judge replies and provider/process mocks do not establish model quality.')

(out/'pipeline-receipt.json').write_text(json.dumps(pipeline,indent=2)+'\n')
