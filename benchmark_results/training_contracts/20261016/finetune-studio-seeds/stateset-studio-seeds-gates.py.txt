"""Repository gates plus serial formatting for this update's changed Python."""
from pathlib import Path
import subprocess
import sys
import black
import isort

for command in [
    [sys.executable,'-m','ruff','check','.'],
    [sys.executable,'scripts/check_repo_hygiene.py'],
    [sys.executable,'scripts/check_release_governance.py'],
    [sys.executable,'scripts/check_api_compatibility.py'],
]:
    print('Checking:', ' '.join(command), flush=True)
    subprocess.run(command,check=True)
for name in ['stateset_agents/cli_remote.py', 'stateset_agents/finetune_workspace.py', 'stateset_agents/finetune_worker.py', 'stateset_agents/tui.py', 'stateset_agents/training/sft.py', 'stateset_agents/remote/job.py', 'stateset_agents/remote/river.py', 'stateset_agents/remote/executor.py', 'stateset_agents/remote/runpod.py', 'stateset_agents/remote/local.py', 'stateset_agents/remote/modal.py', 'stateset_agents/remote/huggingface.py', 'stateset_agents/remote/coreweave.py', 'stateset_agents/remote/nebius.py', 'stateset_agents/finetune_evaluation.py', 'stateset_agents/finetune_evaluation_worker.py', 'stateset_agents/finetune_quality.py', 'tests/unit/test_finetune_seeds.py', 'tests/unit/test_remote_job.py', 'tests/unit/test_finetune_quality.py', 'stateset_agents/finetune_evaluation_tui.py']:
    changed=black.format_file_in_place(Path(name),fast=False,mode=black.Mode(line_length=88,target_versions={black.TargetVersion.PY310}),write_back=black.WriteBack.CHECK)
    assert not changed and isort.check_file(name), name
print('Serial formatting passed for all changed Python files.',flush=True)
