from pathlib import Path
import hashlib
import json
import shutil
import subprocess
import sys
import xml.etree.ElementTree as ET
import zipfile

root = Path('/home/dom/stateset-agents')
source = Path(sys.argv[1])
final_job, prepare_job, *wheel_jobs = sys.argv[2:]
assert len(wheel_jobs) == 3
v = json.loads((source/'verification.json').read_text())
prepared = json.loads((source/'prepare-verification.json').read_text())
assert v['phase']=='complete' and v['passed'] and prepared['phase']=='prepare' and prepared['passed']
package = {p.relative_to(root/'stateset_agents').as_posix(): hashlib.sha256(p.read_bytes()).hexdigest() for p in sorted((root/'stateset_agents').rglob('*.py'))}
tests = {p.relative_to(root).as_posix(): hashlib.sha256(p.read_bytes()).hexdigest() for p in sorted((root/'tests').rglob('*.py'))}
assert package == v['source_sha256'] == prepared['source_sha256']
assert tests == v['test_source_sha256'] == prepared['test_source_sha256']
assert 'Success: no issues found in 425 source files' in (source/'types.log').read_text()
with zipfile.ZipFile(next((source/'dist').glob('*.whl'))) as wheel:
    packed = {n.removeprefix('stateset_agents/'): hashlib.sha256(wheel.read(n)).hexdigest() for n in wheel.namelist() if n.startswith('stateset_agents/') and n.endswith('.py')}
    assert packed == package
    for name in ['stateset_agents/config/presets/experiments/support_recovery_preflight6.json', 'stateset_agents/evaluation/rubrics/support_rubric_v4.md']:
        assert wheel.read(name) == (root/name).read_bytes()

def counts(filename):
    suites = list(ET.parse(source/filename).getroot().iter('testsuite'))
    assert not sum(int(s.get('errors',0))+int(s.get('failures',0)) for s in suites)
    skipped = sum(int(s.get('skipped',0)) for s in suites)
    return dict(passed=sum(int(s.get('tests',0)) for s in suites)-skipped, skipped=skipped)

receipts = []
for handle in [prepare_job, *wheel_jobs, final_job, '6678588e5e5c']:
    record = json.loads(subprocess.check_output(['/home/dom/agent-builds/buildr','show','--json',handle],text=True))
    assert record['state']=='succeeded' and record['exit_code']==0
    receipts.append({key:record.get(key) for key in ['id','state','exit_code','argv','created','started','finished','wait_seconds','elapsed_seconds','reason']})

live = Path('/tmp/stateset-studio-quality-api-live.txt')
draft = Path('/tmp/stateset-studio-quality-api-draft.txt')

(out/'summary.json').write_text(json.dumps(summary,indent=2)+'\n')
(out/'queue-receipts.json').write_text(json.dumps(receipts,indent=2)+'\n')
for name in ['wheel-harness-repair.json','verification.json','prepare-verification.json','frozen-inventory.json','stage-checkpoint.json','regressions.xml','wheel.xml','wheel-shards.json','wheel-shard-1.xml','wheel-shard-2.xml','wheel-shard-3.xml','wheel-shard-1.log','wheel-shard-2.log','wheel-shard-3.log','regressions.log','gates.log','types.log','wheel-origin.log','wheel-test-configuration.json','workflow-lint.log','reward-audit.json','preflight6-plan.json','free-recovery-plan.log','studio-snapshot.log','studio-narrow-snapshot.log','agent-schema.log','agent-models.log','agent-validation.log','agent-invalid-validation.log','agent-data-inspection.log','agent-data-samples.log','agent-workflow.log','operator-fixture.log','agent-quality-import.log','agent-quality-reverify.log','agent-stop-stale.log','agent-stop.log','agent-stop-repeat.log']:
