root=Path('/home/dom/stateset-agents')

def count(name):
    parsed=ET.parse(source/name).getroot(); suites=list(parsed.iter('testsuite'))
    assert not sum(int(s.get(k,0)) for s in suites for k in ['errors','failures','skipped'])
    assert len([c for c in parsed.iter('testcase') if c.get('classname','').startswith('tests.unit.test_finetune_nomination_history')])==19
    return sum(int(s.get('tests',0)) for s in suites)

    value=json.loads(subprocess.check_output(['/home/dom/agent-builds/buildr','show','--json',handle],text=True))
    return {key:value.get(key) for key in ['id','state','exit_code','argv','created','started','finished','wait_seconds','elapsed_seconds','usage','wait_breakdown']}

Selection fixtures use synthetic replies through real scoring/record/replay; broader regressions also exercise tiny CPU models.')

    (out/name).write_text(json.dumps(value,indent=2)+'\n')
for pattern in ['*.log','*.xml']:
