import json
from pathlib import Path
import torch
from peft import LoraConfig, get_peft_model
from transformers import AutoConfig, AutoModelForImageTextToText
from transformers.models.nemotron_h.configuration_nemotron_h import NemotronHConfig
from stateset_agents.training.nemotron_vl import NEMOTRON_VL_MODELS, NEMOTRON_VL_REVISIONS, NEMOTRON_VL_LORA_TARGETS

model_id=NEMOTRON_VL_MODELS['super-vl']
revision=NEMOTRON_VL_REVISIONS['super-vl']
config=AutoConfig.from_pretrained(model_id, revision=revision, trust_remote_code=True)
config.llm_config=NemotronHConfig(vocab_size=256,hidden_size=32,layers_block_type=['full_attention','linear_attention','moe'],num_attention_heads=4,num_key_value_heads=2,head_dim=8,intermediate_size=64,n_routed_experts=4,num_experts_per_tok=2,moe_intermediate_size=16,moe_shared_expert_intermediate_size=16,moe_latent_size=8,use_mamba_kernels=False,mamba_num_heads=8,mamba_head_dim=8,ssm_state_size=8,n_groups=1,chunk_size=8,max_position_embeddings=1024,num_nextn_predict_layers=0,use_cache=False,num_logits_to_keep=0)
config.dtype=torch.float32
config.force_image_size=32
config.vit_hidden_size=32
config.projector_hidden_size=64
config.vision_final_layernorm=True
for key,value in dict(hidden_size=32,num_hidden_layers=1,num_attention_heads=4,image_size=32,cpe_max_size=32,dtype=torch.float32).items():
    setattr(config.vision_config,key,value)
config.video_pruning_rate=0.0
model=AutoModelForImageTextToText.from_config(config,trust_remote_code=True,code_revision=revision)
model=get_peft_model(model,LoraConfig(r=2,lora_alpha=4,target_modules=NEMOTRON_VL_LORA_TARGETS))
model.train()
input_ids=torch.tensor([[4,config.img_context_token_id,7,8]])
labels=input_ids.clone()
labels[:,:2]=-100
pixels=torch.randn(1,3,32,32)
before={n:p.detach().clone() for n,p in model.named_parameters() if p.requires_grad}
optimizer=torch.optim.AdamW([p for p in model.parameters() if p.requires_grad],lr=0.01)
result=model(input_ids=input_ids,attention_mask=torch.ones_like(input_ids),pixel_values=pixels,labels=labels,use_cache=False)
assert torch.isfinite(result.loss)
result.loss.backward()
optimizer.step()
changed=[n for n,p in model.named_parameters() if n in before and not torch.equal(p,before[n])]
assert changed
report={'kind':'actual-pinned-super-vl-code-tiny-random-weight-contract','model':model_id,'revision':revision,'loaded_pretrained_weights':False,'full_size_gpu_training_verified':False,'loss':float(result.loss.detach()),'updated_lora_parameters':changed,'parameters':sum(p.numel() for p in model.parameters())}
Path('/tmp/stateset-super-vl-real-architecture-contract.json').write_text(json.dumps(report,indent=2)+'\n')
print(json.dumps(report),flush=True)
