from pathlib import Path
import fcntl
import hashlib
import json

root = Path('/home/dom/stateset-agents')
staging = Path('/tmp/stateset-ppo-kl-fix')
lock = Path('/tmp/stateset-studio-mobile-source.lock').open('a')
fcntl.flock(lock.fileno(), fcntl.LOCK_EX | fcntl.LOCK_NB)
files = json.loads((staging/'manifest.json').read_text())['files']
for name, info in files.items():
    assert hashlib.sha256((root/name).read_bytes()).hexdigest() == info['base_sha256'], name
for name in files:
    (root/name).write_bytes((staging/name).read_bytes())
print('Applied adaptive KL and reference guard:', ', '.join(files))
