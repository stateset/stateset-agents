def replace_list(path, variable, additions):
    source = path.read_text()
    tree = ast.parse(source)
    node = next(n for n in tree.body if isinstance(n, ast.Assign) and any(isinstance(t,ast.Name) and t.id==variable for t in n.targets))
    old = ast.get_source_segment(source,node)
    values = ast.literal_eval(node.value)
    replacement = list(dict.fromkeys([*additions,*values]))
    path.write_text(source.replace(old,variable+' = '+repr(replacement),1))
